function doGet(e) { var action = e.parameter.action; var ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. ID CARD DATA FETCH if (action === "getIdCardData") { var rollNo = e.parameter.rollNo; var className = e.parameter.className; var sheet = ss.getSheetByName(className) || ss.getSheetByName("All Students"); if (!sheet) { var sheets = ss.getSheets(); for (var s = 0; s < sheets.length; s++) { if (sheets[s].getName().toLowerCase().trim() === String(className).toLowerCase().trim()) { sheet = sheets[s]; break; } } } if (!sheet) { return responseJSON("Not Found"); } var data = sheet.getDataRange().getValues(); for (var i = 2; i < data.length; i++) { var row = data[i]; if (String(row[0]).trim() === String(rollNo).trim()) { var studentObj = { rollNo: row[0], regNo: row[1], class: row[2] || className, section: row[3], name: row[4], father: row[5], mother: row[6], dob: formatDateVal(row[7]), address: row[8], mobile: row[9], photo: row[10], bloodGroup: row[11], joinDate: formatDateVal(row[12]), expireDate: formatDateVal(row[13]) }; return responseJSON(studentObj); } } return responseJSON("Not Found"); } // 2. ATTENDANCE MARK if (action === "markAttendance") { var studentData = JSON.parse(e.parameter.data); var cls = String(studentData.class).trim(); var studentId = String(studentData.id).trim(); var sheet = getSheetCaseInsensitive(ss, cls) || ss.getSheetByName("All Students"); if (!sheet) { return responseJSON({status: "Error", message: "Class Sheet Not Found!"}); } var todayStr = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd/MM/yyyy"); var headers = sheet.getRange(2, 1, 1, sheet.getLastColumn()).getValues()[0]; var dateColIndex = -1; for (var col = 14; col < headers.length; col++) { var headerVal = formatDateVal(headers[col]); if (headerVal === todayStr) { dateColIndex = col + 1; break; } } if (dateColIndex === -1) { dateColIndex = Math.max(headers.length + 1, 15); sheet.getRange(2, dateColIndex).setValue(todayStr).setFontWeight("bold"); } var data = sheet.getDataRange().getValues(); var studentRowIndex = -1; for (var r = 2; r < data.length; r++) { var rowId = String(data[r][1]).trim(); var rowRoll = String(data[r][0]).trim(); var rowName = String(data[r][4]).trim().toLowerCase(); if (rowId === studentId || (rowRoll === String(studentData.roll).trim() && rowName === String(studentData.name).trim().toLowerCase())) { studentRowIndex = r + 1; break; } } if (studentRowIndex === -1) { return responseJSON({status: "Error", message: "Student record not found in class tab!"}); } var currentStatus = sheet.getRange(studentRowIndex, dateColIndex).getValue(); if (String(currentStatus).trim().toLowerCase() === "present") { return responseJSON({status: "Already Marked", name: studentData.name}); } sheet.getRange(studentRowIndex, dateColIndex).setValue("Present"); return responseJSON({status: "Success", name: studentData.name, date: todayStr}); } // 3. DASHBOARD DATA FETCH (Total Students from Column E + Attendance Records) if (action === "getAttendanceDashboard") { var sheets = ss.getSheets(); var allClassTabs = []; var attendanceRecords = []; var totalStudentsSet = new Set(); sheets.forEach(function(sh) { var sName = sh.getName().trim(); var sNameLower = sName.toLowerCase(); if (sNameLower !== "all students" && sNameLower !== "attendance" && sNameLower !== "dashboard") { allClassTabs.push(sName); var values = sh.getDataRange().getValues(); if (values.length > 2) { var headerRow = values[1]; var dateCols = []; for (var c = 14; c < headerRow.length; c++) { var dVal = headerRow[c]; if (dVal) { var formattedD = formatDateVal(dVal); if (formattedD) { dateCols.push({ index: c, date: formattedD }); } } } for (var r = 2; r < values.length; r++) { var rowData = values[r]; // Column E (Index 4) se Student Name count karein agar khali nahi hai if (rowData[4] && String(rowData[4]).trim() !== "") { totalStudentsSet.add(sNameLower + "_" + String(rowData[4]).trim().toLowerCase()); } if (!rowData[0] && !rowData[4]) continue; dateCols.forEach(function(dObj) { var statusVal = rowData[dObj.index]; if (statusVal && String(statusVal).trim() !== "") { attendanceRecords.push({ rollNo: rowData[0], studentId: rowData[1], className: rowData[2] || sName, section: rowData[3], name: rowData[4], date: dObj.date, status: String(statusVal).trim() }); } }); } } } }); return responseJSON({ classList: allClassTabs, records: attendanceRecords, totalStudentsCount: totalStudentsSet.size }); } } function getSheetCaseInsensitive(ss, sheetName) { var sheets = ss.getSheets(); for (var i = 0; i < sheets.length; i++) { if (sheets[i].getName().trim().toLowerCase() === sheetName.trim().toLowerCase()) { return sheets[i]; } } return null; } function formatDateVal(val) { if (!val) return ""; if (val instanceof Date) { return Utilities.formatDate(val, Session.getScriptTimeZone(), "dd/MM/yyyy"); } var strVal = String(val).trim(); if (strVal.includes("T")) { var d = new Date(strVal); if (!isNaN(d.getTime())) { return Utilities.formatDate(d, Session.getScriptTimeZone(), "dd/MM/yyyy"); } } return strVal; } function responseJSON(data) { return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON); }
